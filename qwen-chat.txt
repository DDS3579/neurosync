WHAT
NeuroSync is a privacy-first, neuroscience-informed web app with three parts:
1) Brain Explorer: interactive 3D brain atlas
2) Personal tracker: sleep, focus sessions, mood check-ins, movement, nutrition habits, custom habits
3) Local insight engine + Personal Experiment Lab (rule-based, deterministic)
v0.1 is LOCAL-FIRST: no backend, no accounts, no AI API, no analytics. All data lives in IndexedDB via Dexie.
All data access goes through repositories so auth + sync can be added later by swapping repository internals only.

NON-NEGOTIABLES
1. Scientific honesty. Never claim habits measure or cause brain growth, neural activity, or "brain health".
   Brain-region links are EDUCATIONAL ASSOCIATIONS, never measurements.
   Allowed: "commonly involves", "is associated with". Banned: "your X grew/strengthened", "brain score",
   "brain health score", "boost your IQ", diagnosis language.
2. Privacy. Zero network calls with user data. No analytics/tracking SDKs. Do not claim encryption.
3. Wellbeing over optimization. No shaming copy, no punishing streaks, no calorie/macro targets.
4. Insights are deterministic TypeScript. They require minimum data, always include a caveat,
   and never imply causation.
5. Sleep reference is shown as a reference, never a verdict (teens 8-10h, adults 7h+; user can set their own target).

```typescript
import type { ISODate } from '@/types/domain';

/**
 * Pure descriptive statistics for the insight engine.
 * Empty inputs return null (never NaN); inputs are never mutated.
 * All functions assume finite numbers (Zod validates at the repository boundary).
 */

export function mean(xs: readonly number[]): number | null {
  if (xs.length === 0) return null;
  let sum = 0;
  for (const x of xs) sum += x;
  return sum / xs.length;
}

export function median(xs: readonly number[]): number | null {
  if (xs.length === 0) return null;
  const sorted = [...xs].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  if (sorted.length % 2 === 1) return sorted[mid] ?? null;
  const lo = sorted[mid - 1];
  const hi = sorted[mid];
  return lo === undefined || hi === undefined ? null : (lo + hi) / 2;
}

/** Population standard deviation (descriptive stats over the user's own logs, not a sample estimate). */
export function stdDev(xs: readonly number[]): number | null {
  const m = mean(xs);
  if (m === null) return null;
  let sumSq = 0;
  for (const x of xs) sumSq += (x - m) ** 2;
  return Math.sqrt(sumSq / xs.length);
}

/** Pearson r. Null when n < 5, lengths differ, or either series has (near-)zero variance. */
export function pearson(xs: readonly number[], ys: readonly number[]): number | null {
  if (xs.length !== ys.length || xs.length < 5) return null;
  const mx = mean(xs);
  const my = mean(ys);
  if (mx === null || my === null) return null;
  let sxy = 0;
  let sxx = 0;
  let syy = 0;
  for (const [i, x] of xs.entries()) {
    const y = ys[i];
    if (y === undefined) return null;
    const dx = x - mx;
    const dy = y - my;
    sxy += dx * dy;
    sxx += dx * dx;
    syy += dy * dy;
  }
  // Absolute epsilon also catches fp dust left over from constant series.
  if (sxx < 1e-12 || syy < 1e-12) return null;
  return clamp(sxy / Math.sqrt(sxx * syy), -1, 1); // guard fp overshoot past ±1
}

/** Least-squares slope of ys over xs. Null when n < 2, lengths differ, or x has zero variance. */
export function linearSlope(xs: readonly number[], ys: readonly number[]): number | null {
  if (xs.length !== ys.length || xs.length < 2) return null;
  const mx = mean(xs);
  const my = mean(ys);
  if (mx === null || my === null) return null;
  let sxy = 0;
  let sxx = 0;
  for (const [i, x] of xs.entries()) {
    const y = ys[i];
    if (y === undefined) return null;
    sxy += (x - mx) * (y - my);
    sxx += (x - mx) ** 2;
  }
  if (sxx < 1e-12) return null;
  return sxy / sxx;
}

/** Groups items by their ISODate. Insertion order preserved; items are referenced, never cloned or mutated. */
export function groupByDate<T extends { date: ISODate }>(items: readonly T[]): Map<ISODate, T[]> {
  const out = new Map<ISODate, T[]>();
  for (const item of items) {
    const bucket = out.get(item.date);
    if (bucket) bucket.push(item);
    else out.set(item.date, [item]);
  }
  return out;
}

export interface DateValue {
  date: ISODate;
  value: number;
}

export interface DatePair {
  date: ISODate;
  x: number;
  y: number;
}

/**
 * Inner join of two numeric series on ISODate.
 * Duplicate dates on either side are averaged; output is sorted ascending by date,
 * so the result is independent of input ordering.
 */
export function pairByDate(a: readonly DateValue[], b: readonly DateValue[]): DatePair[] {
  const summarize = (points: readonly DateValue[]): Map<ISODate, { sum: number; count: number }> => {
    const m = new Map<ISODate, { sum: number; count: number }>();
    for (const p of points) {
      const cur = m.get(p.date);
      if (cur) {
        cur.sum += p.value;
        cur.count += 1;
      } else {
        m.set(p.date, { sum: p.value, count: 1 });
      }
    }
    return m;
  };
  const ma = summarize(a);
  const mb = summarize(b);
  const out: DatePair[] = [];
  for (const [date, xa] of ma) {
    const yb = mb.get(date);
    if (yb !== undefined) out.push({ date, x: xa.sum / xa.count, y: yb.sum / yb.count });
  }
  // 'YYYY-MM-DD' strings compare lexicographically in chronological order.
  out.sort((p, q) => (p.date < q.date ? -1 : p.date > q.date ? 1 : 0));
  return out;
}

const MINUTES_PER_DAY = 1440;
const TWO_PI = Math.PI * 2;
const RAD_TO_MINUTES = MINUTES_PER_DAY / TWO_PI;

const wrapMinutes = (m: number): number => ((m % MINUTES_PER_DAY) + MINUTES_PER_DAY) % MINUTES_PER_DAY;

function resultant(minutes: readonly number[]): { c: number; s: number; r: number } | null {
  if (minutes.length === 0) return null;
  let c = 0;
  let s = 0;
  for (const m of minutes) {
    const theta = (wrapMinutes(m) / MINUTES_PER_DAY) * TWO_PI;
    c += Math.cos(theta);
    s += Math.sin(theta);
  }
  c /= minutes.length;
  s /= minutes.length;
  return { c, s, r: Math.hypot(c, s) };
}

/**
 * Circular mean of minutes-of-day, in [0, 1440). Handles midnight wrap:
 * 23:30 (1410) and 00:30 (30) average to 00:00, not 12:00.
 */
export function circularMeanMinutes(minutes: readonly number[]): number | null {
  const res = resultant(minutes);
  if (res === null) return null;
  const angle = Math.atan2(res.s, res.c); // -π..π
  const raw = wrapMinutes(angle * RAD_TO_MINUTES);
  // fp can turn a true-midnight mean into 1439.9999999…; snap it back to 0.
  return MINUTES_PER_DAY - raw < 1e-6 ? 0 : raw;
}

/**
 * Circular standard deviation in minutes (Mardia & Jupp: sqrt(-2 ln R), converted to minutes).
 * Capped at 720 (half a day) so near-antipodal times yield a finite, interpretable spread
 * instead of the formula's unbounded limit as R → 0.
 */
export function circularStdMinutes(minutes: readonly number[]): number | null {
  const res = resultant(minutes);
  if (res === null) return null;
  if (res.r >= 1 - 1e-12) return 0; // identical times (R may be 0.999… in fp)
  if (res.r <= 1e-12) return 720; // opposite clock faces: maximal practical spread
  return Math.min(720, Math.sqrt(-2 * Math.log(res.r)) * RAD_TO_MINUTES);
}

export function clamp(x: number, lo: number, hi: number): number {
  return Math.min(hi, Math.max(lo, x));
}

export function round1(x: number): number {
  return Math.round(x * 10) / 10;
}
```

```typescript
  return items.filter((it) => it.date >= start && it.date <= today);
```

```typescript
import type { Insight, InsightInput, ISODate } from '@/types/domain';
import { RULES, type InsightRule } from './rules';

export { RULES };
export type { InsightRule };

// buildWeeklyReport / analyzeExperiment / generateDemoData live in their own sibling
// modules (weekly.ts, experiments.ts, demo.ts) per the project contract.

const MAX_INSIGHTS = 8;

const KIND_RANK: Record<Insight['kind'], number> = {
  observation: 0,
  reflection: 1,
  milestone: 2,
};

/**
 * Runs every rule against the passed `today` (never the system clock), then orders
 * observations → reflections → milestones. Ties keep rule registration order via an
 * explicit index tiebreak, so ordering is stable on every engine. Caps at 8.
 */
export function computeInsights(input: InsightInput, todayISO: ISODate): Insight[] {
  const found: { order: number; insight: Insight }[] = [];
  RULES.forEach((rule, order) => {
    const insight = rule.run(input, todayISO);
    if (insight !== null) found.push({ order, insight });
  });
  found.sort(
    (a, b) => KIND_RANK[a.insight.kind] - KIND_RANK[b.insight.kind] || a.order - b.order,
  );
  return found.slice(0, MAX_INSIGHTS).map((entry) => entry.insight);
}
```

```typescript
// Soft saturation: ~120 weighted minutes ≈ 0.95 intensity; the curve approaches 1
// asymptotically (1 - exp(-x/k)) so it never hard-clips. k = cap/3.
const SOFT_CAP_WEIGHTED_MIN = 120;
const CURVE_K = SOFT_CAP_WEIGHTED_MIN / 3;

export function activityToRegionIntensity(
  items: readonly { kind: ActivityKind; minutes: number }[],
): Partial<Record<RegionId, number>> {
  const totals = new Map<RegionId, number>();
  for (const item of items) {
    if (!Number.isFinite(item.minutes) || item.minutes <= 0) continue;
    for (const { region, weight } of ACTIVITY_REGION_MAP[item.kind]) {
      if (weight <= 0) continue;
      totals.set(region, (totals.get(region) ?? 0) + item.minutes * weight);
    }
  }
  const out: Partial<Record<RegionId, number>> = {};
  // Fixed iteration order → deterministic key order regardless of input ordering. Zeros omitted.
  for (const { id } of BRAIN_REGIONS) {
    const weighted = totals.get(id);
    if (weighted === undefined || weighted <= 0) continue;
    out[id] = 1 - Math.exp(-weighted / CURVE_K);
  }
  return out;
}
```

```typescript
import type { InsightInput, ISODate, LearningMethod, SleepLog } from '@/types/domain';
import { addDays, format, parseISO } from 'date-fns';
import { groupByDate, mean, round1 } from './stats';

export interface WeeklyReport {
  weekStart: ISODate;
  weekEnd: ISODate;
  sleep: { avgMin: number | null; nightsLogged: number; nightsBelowTarget: number; bedtimeSpreadMin: number | null };
  focus: { totalMin: number; sessions: number; avgFocusRating: number | null; methodsUsed: LearningMethod[] };
  movement: { daysActive: number; totalMin: number };
  wellbeing: { avgMood: number | null; avgStress: number | null; avgEnergy: number | null; avgConcentration: number | null; checkins: number };
  nutrition: { daysLogged: number; regularMealDays: number; avgWater: number | null };
  vsPreviousWeek: Record<string, { delta: number; direction: 'up' | 'down' | 'flat' }>;
  highlights: string[];
  reflectionPrompts: string[];
}

// ---------- private helpers ----------

const isoAdd = (iso: ISODate, days: number): ISODate => format(addDays(parseISO(iso), days), 'yyyy-MM-dd');

// 'YYYY-MM-DD' compares lexicographically in chronological order; windows are inclusive.
function inWindow<T extends { date: ISODate }>(items: readonly T[], start: ISODate, end: ISODate): T[] {
  return items.filter((it) => it.date >= start && it.date <= end);
}

const bedtimeMinutes = (log: SleepLog): number => {
  const d = parseISO(log.bedtime); // offset-less ISO parses as local time (matches local-first logging)
  return d.getHours() * 60 + d.getMinutes();
};

/**
 * Circular SD of minutes-of-day (Mardia), so 23:30 and 00:30 count as 60 min apart.
 * Null with fewer than 2 bedtimes (a spread needs something to spread across).
 * Capped at 720 for interpretability as R → 0.
 */
function bedtimeSpread(bedtimes: readonly number[]): number | null {
  if (bedtimes.length < 2) return null;
  let c = 0;
  let s = 0;
  for (const m of bedtimes) {
    const theta = ((((m % 1440) + 1440) % 1440) / 1440) * 2 * Math.PI;
    c += Math.cos(theta);
    s += Math.sin(theta);
  }
  c /= bedtimes.length;
  s /= bedtimes.length;
  const r = Math.hypot(c, s);
  if (r >= 1 - 1e-12) return 0;
  if (r <= 1e-12) return 720;
  return Math.min(720, Math.sqrt(-2 * Math.log(r)) * (1440 / (2 * Math.PI)));
}

function fmtDur(min: number): string {
  const m = Math.round(min);
  const h = Math.floor(m / 60);
  const r = m % 60;
  if (h === 0) return `${r}m`;
  if (r === 0) return `${h}h`;
  return `${h}h ${r}m`;
}

const plural = (n: number, word: string): string => `${n} ${word}${n === 1 ? '' : 's'}`;

type Sections = Pick<WeeklyReport, 'sleep' | 'focus' | 'movement' | 'wellbeing' | 'nutrition'>;

function computeSections(input: InsightInput, start: ISODate, end: ISODate): Sections {
  const nights = inWindow(input.sleep, start, end);
  const durations = nights.map((n) => n.durationMin);
  const sleepAvg = mean(durations);
  const spread = bedtimeSpread(nights.map(bedtimeMinutes));

  const sessions = inWindow(input.focus, start, end);
  const focusAvg = mean(sessions.map((s) => s.focusRating));

  const movement = inWindow(input.movement, start, end);

  const checkins = inWindow(input.checkins, start, end);
  const avgOf = (pick: (c: (typeof checkins)[number]) => number): number | null => {
    const m = mean(checkins.map(pick));
    return m === null ? null : round1(m);
  };

  // One nutrition entry per day: a later log for the same date supersedes earlier ones.
  const nutritionByDate = new Map<ISODate, (typeof input.nutrition)[number]>();
  for (const log of inWindow(input.nutrition, start, end)) nutritionByDate.set(log.date, log);
  const nutritionDays = [...nutritionByDate.values()];
  const waterAvg = mean(nutritionDays.map((n) => n.waterGlasses));

  return {
    sleep: {
      avgMin: sleepAvg === null ? null : round1(sleepAvg),
      nightsLogged: nights.length,
      nightsBelowTarget: nights.filter((n) => n.durationMin < input.settings.sleepTargetMin).length,
      bedtimeSpreadMin: spread === null ? null : round1(spread),
    },
    focus: {
      totalMin: sessions.reduce((a, s) => a + s.durationMin, 0),
      sessions: sessions.length,
      avgFocusRating: focusAvg === null ? null : round1(focusAvg),
      methodsUsed: [...new Set(sessions.map((s) => s.method))].sort(),
    },
    movement: {
      daysActive: new Set(movement.map((m) => m.date)).size,
      totalMin: movement.reduce((a, m) => a + m.durationMin, 0),
    },
    wellbeing: {
      avgMood: avgOf((c) => c.mood),
      avgStress: avgOf((c) => c.stress),
      avgEnergy: avgOf((c) => c.energy),
      avgConcentration: avgOf((c) => c.concentration),
      checkins: checkins.length,
    },
    nutrition: {
      daysLogged: nutritionDays.length,
      regularMealDays: nutritionDays.filter((n) => n.regularMeals).length,
      avgWater: waterAvg === null ? null : round1(waterAvg),
    },
  };
}

/**
 * Comparison keys and per-metric "flat" tolerances. A key is omitted when either
 * week lacks data (missing ≠ 0); totals/counts are only compared when at least
 * one week actually logged something.
 */
const COMPARISONS: readonly { key: string; tolerance: number; pick: (s: Sections) => number | null; needsData: boolean }[] = [
  { key: 'sleepAvgMin', tolerance: 15, pick: (s) => s.sleep.avgMin, needsData: false },
  { key: 'focusTotalMin', tolerance: 30, pick: (s) => s.focus.totalMin, needsData: true },
  { key: 'movementDaysActive', tolerance: 0, pick: (s) => s.movement.daysActive, needsData: true },
  { key: 'avgMood', tolerance: 0.3, pick: (s) => s.wellbeing.avgMood, needsData: false },
  { key: 'avgStress', tolerance: 0.3, pick: (s) => s.wellbeing.avgStress, needsData: false },
  { key: 'avgEnergy', tolerance: 0.3, pick: (s) => s.wellbeing.avgEnergy, needsData: false },
  { key: 'avgConcentration', tolerance: 0.3, pick: (s) => s.wellbeing.avgConcentration, needsData: false },
  { key: 'avgWater', tolerance: 1, pick: (s) => s.nutrition.avgWater, needsData: false },
];

function bestFocusDay(input: InsightInput, start: ISODate, end: ISODate): { date: ISODate; avg: number } | null {
  const perDay: { date: ISODate; avg: number }[] = [];
  for (const [date, list] of groupByDate(inWindow(input.focus, start, end))) {
    const m = mean(list.map((s) => s.focusRating));
    if (m !== null) perDay.push({ date, avg: m });
  }
  perDay.sort((a, b) => (a.date < b.date ? -1 : a.date > b.date ? 1 : 0));
  let best: { date: ISODate; avg: number } | null = null;
  for (const day of perDay) if (best === null || day.avg > best.avg) best = day; // strict > keeps earliest on ties
  return best;
}

// ---------- public API ----------

export function buildWeeklyReport(input: InsightInput, weekStartISO: ISODate): WeeklyReport {
  const weekStart = weekStartISO;
  const weekEnd = isoAdd(weekStartISO, 6);
  const current = computeSections(input, weekStart, weekEnd);
  const previous = computeSections(input, isoAdd(weekStartISO, -7), isoAdd(weekStartISO, -1));

  const vsPreviousWeek: WeeklyReport['vsPreviousWeek'] = {};
  for (const { key, tolerance, pick, needsData } of COMPARISONS) {
    const cur = pick(current);
    const prev = pick(previous);
    if (cur === null || prev === null) continue;
    if (needsData && cur === 0 && prev === 0) continue;
    const raw = cur - prev;
    vsPreviousWeek[key] = {
      delta: round1(raw),
      direction: Math.abs(raw) <= tolerance + 1e-9 ? 'flat' : raw > 0 ? 'up' : 'down',
    };
  }

  const highlights: string[] = [];
  const { sleep, focus, movement, wellbeing } = current;
  if (sleep.nightsLogged > 0 && sleep.avgMin !== null) {
    highlights.push(`You logged ${plural(sleep.nightsLogged, 'night')} of sleep, averaging ${fmtDur(sleep.avgMin)}.`);
  }
  if (focus.sessions > 0) {
    const variety = focus.methodsUsed.length >= 3 ? `, spanning ${focus.methodsUsed.length} different methods` : '';
    highlights.push(`${plural(focus.sessions, 'focus session')} added up to ${fmtDur(focus.totalMin)} of practice${variety}.`);
  }
  const best = bestFocusDay(input, weekStart, weekEnd);
  if (best !== null) {
    highlights.push(`Your best-focus day was ${format(parseISO(best.date), 'EEEE')} (${round1(best.avg)}/5 average).`);
  }
  if (movement.daysActive >= 4) {
    highlights.push(`Movement showed up on ${movement.daysActive} days — a steady rhythm.`);
  } else if (movement.daysActive > 0) {
    highlights.push(`You moved on ${plural(movement.daysActive, 'day')} this week.`);
  }
  if (wellbeing.checkins >= 5) {
    highlights.push(`${wellbeing.checkins} check-ins — a week of noticing how you are.`);
  }
  if (highlights.length === 0) {
    highlights.push('A quiet week in the logs — they only show what was recorded, not everything you lived.');
  }

  const prompts: string[] = [];
  if (focus.sessions > 0) prompts.push('What made your best-focus day different?');
  if (sleep.nightsLogged > 0) prompts.push('What did the evening before your longest night look like?');
  if (movement.daysActive > 0) prompts.push('How did the days you moved feel in your body and mood?');
  if (wellbeing.checkins > 0) prompts.push('Which check-in surprised you, and what was happening around it?');
  prompts.push('What is one small thing from this week you would like to keep?');
  prompts.push('What would make next week feel a little easier?');

  return {
```

```typescript
import type { Experiment, ExperimentMetric, InsightInput, ISODate } from '@/types/domain';
import { groupByDate, mean, round1, stdDev } from './stats';

export interface ExperimentResult {
  baseline: { n: number; mean: number | null };
  intervention: { n: number; mean: number | null };
  difference: number | null;
  sufficientData: boolean;
  direction: 'higher' | 'lower' | 'similar' | 'unknown';
  summary: string;
  caveats: string[];
  effectSizeNote?: string;
}

const MIN_DAYS = 5;
// On the 1–5 rating scale, differences within ±0.3 are described as "similar".
const SIMILAR_BAND = 0.3;

const METRIC_LABELS: Record<ExperimentMetric, string> = {
  concentration: 'concentration',
  mood: 'mood',
  energy: 'energy',
  stress: 'stress',
  focusRating: 'session focus',
};

const plural = (n: number, word: string): string => `${n} ${word}${n === 1 ? '' : 's'}`;

/** One value per logged day: check-in field, or the day's mean session focus rating. */
function metricByDate(exp: Experiment, input: InsightInput): Map<ISODate, number> {
  const out = new Map<ISODate, number>();
  if (exp.metric === 'focusRating') {
    for (const [date, list] of groupByDate(input.focus)) {
      const m = mean(list.map((s) => s.focusRating));
      if (m !== null) out.set(date, m);
    }
  } else {
    const key = exp.metric; // 'concentration' | 'mood' | 'energy' | 'stress'
    for (const [date, list] of groupByDate(input.checkins)) {
      const m = mean(list.map((c) => c[key])); // several check-ins on one day are averaged
      if (m !== null) out.set(date, m);
    }
  }
  return out;
}

function periodValues(series: Map<ISODate, number>, start: ISODate, end: ISODate): number[] {
  const vals: number[] = [];
  for (const [date, value] of series) {
    if (date >= start && date <= end) vals.push(value); // inclusive ranges, ISO string order
  }
  return vals;
}

export function analyzeExperiment(exp: Experiment, input: InsightInput): ExperimentResult {
  const series = metricByDate(exp, input);
  const bVals = periodValues(series, exp.baselineStart, exp.baselineEnd);
  const iVals = periodValues(series, exp.interventionStart, exp.interventionEnd);
  const bMean = mean(bVals);
  const iMean = mean(iVals);
  const sufficientData = bVals.length >= MIN_DAYS && iVals.length >= MIN_DAYS;
  const difference = bMean !== null && iMean !== null ? Math.round((iMean - bMean) * 100) / 100 : null;

  // Direction is only stated once both periods clear the minimum — otherwise 'unknown'.
  let direction: ExperimentResult['direction'] = 'unknown';
  if (sufficientData && difference !== null) {
    direction = difference > SIMILAR_BAND ? 'higher' : difference < -SIMILAR_BAND ? 'lower' : 'similar';
  }

  const label = METRIC_LABELS[exp.metric];
  let summary: string;
  if (bMean === null || iMean === null) {
    const where = bMean === null && iMean === null ? 'either period' : 'one of the periods';
    summary = `No ${label} data was logged in ${where} yet, so there is nothing to compare. Baseline days: ${bVals.length}. Intervention days: ${iVals.length}.`;
  } else if (!sufficientData) {
    summary = `So far, ${plural(bVals.length, 'baseline day')} average ${round1(bMean)}/5 and ${plural(iVals.length, 'intervention day')} average ${round1(iMean)}/5 for ${label}. At least ${MIN_DAYS} days in each period are needed before a comparison is shown — these numbers are provisional.`;
  } else {
    // Neutral wording for every metric: "lower stress rating" is a description, never a verdict.
    const phrase = direction === 'similar' ? `similar ${label} rating (within ±${SIMILAR_BAND})` : `${direction} ${label} rating`;
    const sign = difference !== null && difference >= 0 ? '+' : '';
    summary = `Across ${plural(bVals.length, 'baseline day')} and ${plural(iVals.length, 'intervention day')}, the average ${label} rating went from ${round1(bMean)}/5 to ${round1(iMean)}/5 — a ${phrase} (difference ${sign}${difference}). This describes your logs; it does not show the change caused the difference.`;
  }

  const caveats: string[] = [
    'This comparison does not establish causation.',
    'Self-reported ratings are subjective and can shift with mood and context.',
    'Other things changed during the period too — sleep, workload, season, life — and any of them could play a part.',
  ];
  if (!sufficientData) {
    caveats.push(`With fewer than ${MIN_DAYS} days in a period, averages are especially noisy.`);
  }

  let effectSizeNote: string | undefined;
  if (sufficientData && difference !== null) {
    const sb = stdDev(bVals);
    const si = stdDev(iVals);
    if (sb !== null && si !== null) {
      const pooled = Math.sqrt((sb * sb + si * si) / 2);
      if (pooled > 1e-9) {
        effectSizeNote = `For scale: the difference is about ${round1(Math.abs(difference) / pooled)} times your typical day-to-day variation — a rough sense of size, not a verdict.`;
      }
    }
  }

  return {
    baseline: { n: bVals.length, mean: bMean === null ? null : round1(bMean) },
    intervention: { n: iVals.length, mean: iMean === null ? null : round1(iMean) },
    difference,
    sufficientData,
    direction,
    summary,
    caveats,
    ...(effectSizeNote !== undefined ? { effectSizeNote } : {}),
  };
}
```

```typescript
import type {
  DailyCheckin, FocusSession, Habit, HabitLog, InsightInput, ISODate,
  LearningMethod, MovementLog, NutritionLog, Rating, SleepLog, UserSettings,
} from '@/types/domain';
import { addMinutes, format, getDay, startOfDay, subDays } from 'date-fns';
import { clamp } from './stats';

/** Mulberry32: tiny, fast, deterministic 32-bit PRNG. */
function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

const METHODS: readonly LearningMethod[] = [
  'active_recall', 'spaced_repetition', 'problem_solving', 'deep_reading', 'skill_practice', 'teaching',
];
const SUBJECTS = ['Math', 'Reading', 'Code', 'Language', 'Music', 'Writing'] as const;
const MOVEMENT_TYPES: readonly MovementLog['type'][] = ['walk', 'run', 'cycle', 'gym', 'stretch', 'sport'];
const SESSION_DURATIONS = [25, 30, 45, 50, 60, 90] as const;

const HABIT_DEFS: readonly { name: string; icon: string }[] = [
  { name: 'Read 10 pages', icon: '📖' },
  { name: 'Morning stretch', icon: '🤸' },
  { name: 'Walk outside', icon: '🚶' },
  { name: 'Evening journal', icon: '✍️' },
];

const toRating = (x: number): Rating => clamp(Math.round(x), 1, 5) as Rating;
const wrapDay = (m: number): number => ((m % 1440) + 1440) % 1440;

/**
 * Deterministic demo dataset ending on `today` (calendar day; time-of-day is normalized away).
 * Same (days, seed, calendar day) → identical output, including ids and timestamps.
 * Concentration is mildly correlated (r ≈ 0.4) with the sleep that ended that morning.
 */
export function generateDemoData(days: number, seed: number, today: Date = new Date()): InsightInput {
  const rand = mulberry32(seed);
  const gaussian = (): number => {
    const u1 = Math.max(rand(), 1e-12); // Box–Muller; guard log(0)
    const u2 = rand();
    return Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
  };
  const pick = <T>(arr: readonly T[]): T => {
    const idx = Math.min(arr.length - 1, Math.floor(rand() * arr.length));
    const value = arr[idx];
    if (value === undefined) throw new Error('pick from empty array');
    return value;
  };

  const totalDays = Math.max(0, Math.floor(days));
  const end = startOfDay(today);
  const todayISO = format(end, 'yyyy-MM-dd');

  const settings: UserSettings = {
    id: 'demo-settings', createdAt: `${todayISO}T08:00:00`, updatedAt: `${todayISO}T08:00:00`,
    displayName: 'Demo', sleepTargetMin: 480, focusDefaultMin: 25, breakDefaultMin: 5,
    onboarded: true, schemaVersion: 1,
  };
  const habits: Habit[] = HABIT_DEFS.map((h, i) => ({
    id: `demo-habit-${i + 1}`, createdAt: `${todayISO}T08:00:00`, updatedAt: `${todayISO}T08:00:00`,
    name: h.name, icon: h.icon, archived: false,
  }));

  const sleep: SleepLog[] = [];
  const checkins: DailyCheckin[] = [];
  const focus: FocusSession[] = [];
  const movement: MovementLog[] = [];
  const nutrition: NutritionLog[] = [];
  const habitLogs: HabitLog[] = [];

  // Fixed call order for rand() → determinism. Oldest day first, so arrays are chronological.
  for (let k = totalDays - 1; k >= 0; k--) {
    const dateObj = subDays(end, k);
    const date: ISODate = format(dateObj, 'yyyy-MM-dd');
    const dow = getDay(dateObj);
    const weekend = dow === 0 || dow === 6; // wake-up date on Sat/Sun: sleep-in + later bedtime

    // --- sleep (every day): N(7.2h, 0.9h), weekend drift, clamped to 5–10h ---
    const zSleep = gaussian();
    const durationMin = Math.round(clamp(432 + 54 * zSleep + (weekend ? 20 : 0), 300, 600));
    const bedNoise = gaussian();
    const bedMin = wrapDay(Math.round(1395 + 35 * bedNoise + (weekend ? 45 : 0))); // ~23:15 base
    // bedMin is always evening (>21:00) or early night (<03:00), so wake lands on `date`.
    const bedBase = bedMin >= 720 ? startOfDay(subDays(dateObj, 1)) : dateObj;
    const bedDate = addMinutes(bedBase, bedMin);
    const wakeDate = addMinutes(bedDate, durationMin);
    const bedtime = format(bedDate, "yyyy-MM-dd'T'HH:mm:ss");
    const wakeTime = format(wakeDate, "yyyy-MM-dd'T'HH:mm:ss");
    const quality = toRating(3 + (durationMin - 432) / 90 + (rand() - 0.5));
    const rested = toRating(3 + (durationMin - 432) / 90 + (rand() - 0.5));
    const interruptions = rand() < 0.7 ? 0 : rand() < 0.8 ? 1 : 2;
    sleep.push({ id: `demo-sleep-${date}`, createdAt: wakeTime, updatedAt: wakeTime, date, bedtime, wakeTime, durationMin, quality, rested, interruptions });

    // --- check-in (~85%): concentration shares 0.45 of its variance with zSleep ---
    if (rand() < 0.85) {
      const zc = gaussian();
      const concentration = toRating(3 + 0.45 * zSleep + 0.893 * zc);
      const mood = toRating(3 + 0.3 * zSleep + 0.9 * gaussian());
      const stress = toRating(3 - 0.25 * zSleep + 0.9 * gaussian());
      const energy = toRating(3 + 0.35 * zSleep + 0.9 * gaussian());
      const motivation = toRating(3 + 0.9 * gaussian());
      const outdoorMin = Math.round(rand() * 70);
      const at = `${date}T21:00:00`;
      checkins.push({ id: `demo-checkin-${date}`, createdAt: at, updatedAt: at, date, mood, stress, energy, concentration, motivation, outdoorMin });
    }

    // --- focus sessions (~70% of days, 1–2 per day) ---
    if (rand() < 0.7) {
      const nSessions = rand() < 0.3 ? 2 : 1;
      for (let i = 0; i < nSessions; i++) {
        const dur = pick(SESSION_DURATIONS);
        const method = pick(METHODS);
        const subject = pick(SUBJECTS);
        const focusRating = toRating(3 + 0.5 * zSleep + 0.8 * gaussian());
        const difficulty = (2 + Math.floor(rand() * 3)) as Rating;
        const distractions = Math.floor(rand() * (dur > 60 ? 4 : 3));
        const startedAt = `${date}T${i === 0 ? '09' : '16'}:00:00`;
        focus.push({ id: `demo-focus-${date}-${i + 1}`, createdAt: startedAt, updatedAt: startedAt, date, startedAt, durationMin: dur, plannedMin: dur, subject, method, focusRating, difficulty, distractions });
      }
    }

    // --- movement (~4 days/week) ---
    if (rand() < 4 / 7) {
      const type = pick(MOVEMENT_TYPES);
      const minutes = 20 + Math.floor(rand() * 41);
      const effort = (2 + Math.floor(rand() * 3)) as Rating;
      const at = `${date}T18:00:00`;
      movement.push({ id: `demo-movement-${date}`, createdAt: at, updatedAt: at, date, type, durationMin: minutes, effort });
    }

    // --- nutrition (~70%) ---
    if (rand() < 0.7) {
      const regularMeals = rand() < 0.75;
      const fruitVeg = rand() < 0.7;
      const varied = rand() < 0.65;
      const waterGlasses = 4 + Math.floor(rand() * 6);
      const at = `${date}T20:00:00`;
      nutrition.push({ id: `demo-nutrition-${date}`, createdAt: at, updatedAt: at, date, regularMeals, fruitVeg, varied, waterGlasses });
    }

    // --- habits: a done-log on ~60% of days per habit ---
    for (let h = 0; h < habits.length; h++) {
      if (rand() < 0.6) {
        const habit = habits[h];
        if (habit === undefined) continue;
        const at = `${date}T21:30:00`;
        habitLogs.push({ id: `demo-habitlog-${h + 1}-${date}`, createdAt: at, updatedAt: at, date, habitId: habit.id, done: true });
      }
    }
  }

  return { settings, sleep, checkins, focus, movement, nutrition, habits, habitLogs };
}
```

```typescript
  Array.from({ length: count }, (_, i) => `2026-11-${String(start + i).padStart(2, '0')}`);
const REQUIRED_CAVEATS = [/causation/i, /self-reported/i, /other things changed/i];

describe('analyzeExperiment — insufficient data', () => {

    expect(r.intervention).toEqual({ n: 0, mean: null });
    expect(r.difference).toBeNull();
    expect(r.sufficientData).toBe(false);
    expect(r.direction).toBe('unknown');
    for (const pattern of REQUIRED_CAVEATS) {
      expect(r.caveats.some((c) => pattern.test(c))).toBe(true);
    }
  });

  it('stays unknown with 4 baseline days even when the difference looks large', () => {
    const checkins = [
      ...days(1, 4).map((d) => checkin(d, { concentration: 2 })),

    expect(r.effectSizeNote).toBeUndefined();

  it('boundary: exactly 5 days in each period is sufficient; 5 and 4 is not', () => {

    const fiveI = days(8, 5).map((d) => checkin(d));

describe('analyzeExperiment — directions and wording', () => {

    const checkins = [...days(1, 5).map((d) => checkin(d, { concentration: 2 })), ...days(8, 5).map((d) => checkin(d, { concentration: 4 }))];
    const r = analyzeExperiment(experiment(), input({ checkins }));

    const r = analyzeExperiment(experiment({ metric: 'stress' }), input({ checkins }));

    const freeze = (v: unknown): void => {
      if (v !== null && typeof v === 'object') {
        Object.freeze(v);
        for (const x of Object.values(v as Record<string, unknown>)) freeze(x);
      }
    };
    freeze(data);
    expect(analyzeExperiment(experiment(), data)).toEqual(expected);
  });
});
```
